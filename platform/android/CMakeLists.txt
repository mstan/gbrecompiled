cmake_minimum_required(VERSION 3.22)
project(gbrecompiled_android C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS gbc_native=4)
set(CMAKE_JOB_POOL_COMPILE gbc_native)
set(CMAKE_JOB_POOL_LINK gbc_native)
if(NOT ANDROID OR NOT EXISTS "${SDL2_SOURCE_DIR}/CMakeLists.txt" OR
   NOT EXISTS "${GBC_GAME_ROOT}/${GBC_GENERATED_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Android needs SDL2_SOURCE_DIR and a generated GBC_GAME_ROOT/GBC_GENERATED_DIR")
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDL2_SOURCE_DIR}" SDL)
set(GBRECOMP_RECOMP_UI ON CACHE BOOL "" FORCE)
set(RECOMP_UI_ENABLE_MODS ON CACHE BOOL "" FORCE)
add_subdirectory("${GBC_GAME_ROOT}/${GBC_GENERATED_DIR}" game)
